CREATE DATABASE jadwalbegie;
USE jadwalbegie;

CREATE TABLE IF NOT EXISTS pengguna (
id_pengguna INT PRIMARY KEY NOT NULL,
nama VARCHAR(100) NOT NULL
);

CREATE TABLE IF NOT EXISTS matakulyeah (
id_mk INT PRIMARY KEY NOT NULL,
nama_mk VARCHAR(100) NOT NULL,
id_pengguna INT, 
CONSTRAINT fk_pengguna
FOREIGN KEY (id_pengguna) REFERENCES pengguna (id_pengguna)
);

CREATE TABLE IF NOT EXISTS jadwal (
hari VARCHAR(10) NOT NULL,
jam TIME NOT NULL, 
ruangan VARCHAR(50),
id_mk INT,
CONSTRAINT fk_matakulyeah
FOREIGN KEY (id_mk) REFERENCES matakulyeah (id_mk)
);
